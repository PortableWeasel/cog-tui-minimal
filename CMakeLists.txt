cmake_minimum_required(VERSION 3.16)
project(cog_tui_minimal VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TUI_SANITIZERS "Enable address/undefined sanitizers" OFF)
find_package(cog-provider REQUIRED CONFIG)
find_package(Threads REQUIRED)
foreach(component config toolkit orchestrator cli)
 include("${cog-provider_DIR}/../cog/cog-${component}-targets.cmake")
endforeach()
find_path(READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
find_library(READLINE_LIBRARY readline REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Werror)
if(TUI_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(tui-md4c STATIC vendor/md4c/md4c.c)
# Keep the pinned upstream source unmodified; strict warnings apply to our adapter.
target_compile_options(tui-md4c PRIVATE -w)
target_include_directories(tui-md4c PUBLIC vendor/md4c)
add_library(tui STATIC src/settings.c src/runtime.c src/markdown.c src/screen.c src/transcript.c)
target_include_directories(tui PUBLIC include)
target_link_libraries(tui PUBLIC cog::cli PRIVATE CURL::libcurl tui-md4c)
add_executable(cog-tui src/main.c src/app.c)
target_include_directories(cog-tui PRIVATE ${READLINE_INCLUDE_DIR})
target_link_libraries(cog-tui PRIVATE tui ${READLINE_LIBRARY})
# Deliberate opt-in only: no live test is registered with CTest.
add_executable(live-acceptance EXCLUDE_FROM_ALL tests/live_acceptance.c)
target_link_libraries(live-acceptance PRIVATE tui)
include(CTest)
if(BUILD_TESTING)
 add_executable(transcript-test tests/transcript_test.c)
 target_link_libraries(transcript-test PRIVATE tui)
 add_test(NAME transcript COMMAND transcript-test)
 add_executable(screen-test tests/screen_test.c)
 target_link_libraries(screen-test PRIVATE tui util)
 add_test(NAME screen COMMAND screen-test)
 set_tests_properties(screen PROPERTIES TIMEOUT 30)
 add_executable(markdown-test tests/markdown_test.c)
 target_link_libraries(markdown-test PRIVATE tui)
 add_test(NAME markdown COMMAND markdown-test)
 add_executable(markdown-fuzz-test tests/markdown_fuzz_test.c)
 target_link_libraries(markdown-fuzz-test PRIVATE tui)
 add_test(NAME markdown-fuzz COMMAND markdown-fuzz-test)
 if(UNIX AND NOT APPLE)
  add_executable(markdown-alloc-test tests/markdown_alloc_test.c)
  target_link_libraries(markdown-alloc-test PRIVATE tui)
  target_link_options(markdown-alloc-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
  add_test(NAME markdown-alloc COMMAND markdown-alloc-test)
  set_tests_properties(markdown-alloc PROPERTIES TIMEOUT 30)
 endif()
 set_tests_properties(markdown markdown-fuzz PROPERTIES TIMEOUT 30)
 add_executable(settings-test tests/settings_test.c)
 target_link_libraries(settings-test PRIVATE tui)
 add_test(NAME settings COMMAND settings-test ${CMAKE_CURRENT_BINARY_DIR})
 add_executable(runtime-test tests/runtime_test.c)
 target_link_libraries(runtime-test PRIVATE tui)
 add_test(NAME runtime COMMAND runtime-test)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME screen-cells COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/screen_cells_test.py $<TARGET_FILE:screen-test>)
 set_tests_properties(screen-cells PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
 add_test(NAME pi-launcher COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/pi_launcher_test.py
  ${CMAKE_CURRENT_SOURCE_DIR}/scripts/cog-tui-pi ${CMAKE_CURRENT_BINARY_DIR})
 add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py $<TARGET_FILE:cog-tui>)
 add_test(NAME streaming-screen COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/streaming_screen_test.py $<TARGET_FILE:cog-tui>)
 set_tests_properties(streaming-screen PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
 set_tests_properties(settings runtime terminal PROPERTIES TIMEOUT 30)
 foreach(scenario first-run credential-recovery provider-failure eof sigint-idle sigterm-active sigint-active output-hangup save-retry)
  add_test(NAME failure-${scenario} COMMAND ${Python3_EXECUTABLE}
   ${CMAKE_CURRENT_SOURCE_DIR}/tests/failure_test.py $<TARGET_FILE:cog-tui> ${scenario})
  set_tests_properties(failure-${scenario} PROPERTIES TIMEOUT 15)
 endforeach()
endif()
set_target_properties(cog-tui PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
foreach(header settings runtime app markdown screen transcript)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.c" "#include <cog_tui/${header}.h>\n")
 add_library(header-${header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.c")
 target_link_libraries(header-${header} PRIVATE tui)
endforeach()
install(TARGETS cog-tui RUNTIME DESTINATION bin)
